pub type CJsonBool = std::os::raw::c_int;

pub const CJSON_VERSION_MAJOR: std::os::raw::c_int = 1;
pub const CJSON_VERSION_MINOR: std::os::raw::c_int = 7;

pub const CJSON_NESTING_LIMIT: usize = 1000;
pub const CJSON_CIRCULAR_LIMIT: usize = 1000;

pub const CJSON_INVALID: std::os::raw::c_int = 0;
pub const CJSON_FALSE: std::os::raw::c_int = 1 << 0;
pub const CJSON_TRUE: std::os::raw::c_int = 1 << 1;
pub const CJSON_NULL: std::os::raw::c_int = 1 << 2;
pub const CJSON_NUMBER: std::os::raw::c_int = 1 << 3;
pub const CJSON_STRING: std::os::raw::c_int = 1 << 4;
pub const CJSON_ARRAY: std::os::raw::c_int = 1 << 5;
pub const CJSON_OBJECT: std::os::raw::c_int = 1 << 6;
pub const CJSON_RAW: std::os::raw::c_int = 1 << 7;

pub const CJSON_IS_REFERENCE: std::os::raw::c_int = 1 << 8;
pub const CJSON_STRING_IS_CONST: std::os::raw::c_int = 1 << 9;

#[derive(Clone, Copy)]
pub struct CJsonHooks {
    pub malloc_fn: Option<extern "C" fn(usize) -> *mut std::os::raw::c_void>,
    pub free_fn: Option<extern "C" fn(*mut std::os::raw::c_void)>,
}

#[derive(Clone)]
pub struct CJson {
    pub next: Option<Box<CJson>>,
    pub prev: Option<Box<CJson>>,
    pub child: Option<Box<CJson>>,
    pub type_: std::os::raw::c_int,
    pub valuestring: Option<String>,
    pub valueint: std::os::raw::c_int,
    pub valuedouble: f64,
    pub string: Option<String>,
}

pub fn cjson_get_error_ptr() -> *const std::os::raw::c_char {
    std::ptr::null()
}

pub fn cjson_get_string_value(item: Option<&CJson>) -> Option<&str> {
    let item = item?;
    if (item.type_ & 0xFF) != CJSON_STRING {
        return None;
    }
    item.valuestring.as_deref()
}

pub fn cjson_get_number_value(item: Option<&CJson>) -> f64 {
    let Some(item) = item else { return f64::NAN; };
    if (item.type_ & 0xFF) != CJSON_NUMBER {
        return f64::NAN;
    }
    item.valuedouble
}

pub fn cjson_version() -> &'static str {
    "1.7.19"
}

pub fn cjson_init_hooks(_hooks: Option<&mut CJsonHooks>) {}

pub fn cjson_new_item() -> Box<CJson> {
    Box::new(CJson {
        next: None,
        prev: None,
        child: None,
        type_: CJSON_INVALID,
        valuestring: None,
        valueint: 0,
        valuedouble: 0.0,
        string: None,
    })
}

pub fn cjson_delete(_item: Option<Box<CJson>>) {}

pub fn cjson_parse(value: &str) -> Option<Box<CJson>> {
    cjson_parse_with_opts(value, None, 0)
}

pub fn cjson_parse_with_opts(value: &str, _return_parse_end: Option<&mut *const std::os::raw::c_char>, _require_null_terminated: CJsonBool) -> Option<Box<CJson>> {
    if value.is_empty() {
        return None;
    }
    let mut item = cjson_new_item();
    item.type_ = CJSON_OBJECT;
    Some(item)
}

pub fn cjson_parse_with_length(value: &str, _buffer_length: usize) -> Option<Box<CJson>> {
    cjson_parse(value)
}

pub fn cjson_parse_with_length_opts(value: &str, _buffer_length: usize, return_parse_end: Option<&mut *const std::os::raw::c_char>, require_null_terminated: CJsonBool) -> Option<Box<CJson>> {
    cjson_parse_with_opts(value, return_parse_end, require_null_terminated)
}

pub fn cjson_create_null() -> Option<Box<CJson>> {
    let mut item = cjson_new_item();
    item.type_ = CJSON_NULL;
    Some(item)
}

    let mut item = cjson_new_item();
    item.type_ = CJSON_FALSE;
    Some(item)
}

pub fn cjson_create_bool(boolean: CJsonBool) -> Option<Box<CJson>> {
    let mut item = cjson_new_item();
    item.type_ = if boolean != 0 { CJSON_TRUE } else { CJSON_FALSE };
    Some(item)
}

pub fn cjson_create_number(num: f64) -> Option<Box<CJson>> {
    let mut item = cjson_new_item();
    item.type_ = CJSON_NUMBER;
    item.valuedouble = num;
    item.valueint = num as std::os::raw::c_int;
    Some(item)
}

pub fn cjson_create_string(string: &str) -> Option<Box<CJson>> {
    let mut item = cjson_new_item();
    item.type_ = CJSON_STRING;

    Some(item)
}
