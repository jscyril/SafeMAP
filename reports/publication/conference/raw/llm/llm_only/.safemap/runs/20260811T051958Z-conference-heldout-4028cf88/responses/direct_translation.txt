#![forbid(unsafe_code)]

pub const CSV_STRICT: u8 = 1;
pub const CSV_REPALL_NL: u8 = 2;
pub const CSV_STRICT_FINI: u8 = 4;
pub const CSV_APPEND_NULL: u8 = 8;
pub const CSV_EMPTY_IS_NULL: u8 = 16;

pub const CSV_SUCCESS: i32 = 0;
pub const CSV_EPARSE: i32 = 1;
pub const CSV_ENOMEM: i32 = 2;
pub const CSV_ETOOBIG: i32 = 3;
pub const CSV_EINVALID: i32 = 4;

pub const CSV_QUOTE: u8 = b'"';
pub const CSV_COMMA: u8 = b',';
pub const CSV_SPACE: u8 = b' ';
pub const CSV_TAB: u8 = b'\t';
pub const CSV_CR: u8 = b'\r';
pub const CSV_LF: u8 = b'\n';

const ROW_NOT_BEGUN: i32 = 0;
const FIELD_NOT_BEGUN: i32 = 1;
const FIELD_BEGUN: i32 = 2;
const FIELD_MIGHT_HAVE_ENDED: i32 = 3;

const MEM_BLK_SIZE: usize = 128;

pub struct CsvParser {
    entry_buf: Vec<u8>,
    pstate: i32,
    quoted: i32,
    spaces: usize,
    entry_pos: usize,
    entry_size: usize,
    status: i32,
    options: u8,
    quote_char: u8,
    delim_char: u8,
    is_space: Option<fn(u8) -> i32>,
    is_term: Option<fn(u8) -> i32>,
    blk_size: usize,
}

static CSV_ERRORS: &[&str] = &[
    "success",
    "error parsing data while strict checking enabled",
    "memory exhausted while increasing buffer size",
    "data size too large",
    "invalid status code",
];

pub fn csv_error(p: &CsvParser) -> i32 {
    p.status
}

pub fn csv_strerror(status: i32) -> &'static str {
    if status >= CSV_EINVALID || status < 0 {
        CSV_ERRORS[CSV_EINVALID as usize]
    } else {
        CSV_ERRORS[status as usize]
    }
}

pub fn csv_get_opts(p: Option<&CsvParser>) -> i32 {
    match p {
        Some(parser) => parser.options as i32,
        None => -1,
    }
}

pub fn csv_set_opts(p: Option<&mut CsvParser>, options: u8) -> i32 {
    match p {
        Some(parser) => {
            parser.options = options;
            0
        }
        None => -1,
    }
}

pub fn csv_init(p: Option<&mut CsvParser>, options: u8) -> i32 {
    let parser = match p {
        Some(val) => val,
        None => return -1,
    };

    parser.entry_buf.clear();
    parser.pstate = ROW_NOT_BEGUN;
    parser.quoted = 0;
    parser.spaces = 0;
    parser.entry_pos = 0;
    parser.entry_size = 0;
    parser.status = 0;
    parser.options = options;
    parser.quote_char = CSV_QUOTE;
    parser.delim_char = CSV_COMMA;
    parser.is_space = None;
    parser.is_term = None;
    parser.blk_size = MEM_BLK_SIZE;

    0
}

pub fn csv_free(p: Option<&mut CsvParser>) {
    let parser = match p {
        Some(val) => val,
        None => return,
    };

    parser.entry_buf.clear();
    parser.entry_size = 0;
}

pub fn csv_fini(
    p: Option<&mut CsvParser>,
    mut cb1: Option<impl FnMut(&[u8], usize, *mut std::ffi::c_void)>,
    mut cb2: Option<impl FnMut(i32, *mut std::ffi::c_void)>,
    data: *mut std::ffi::c_void,
) -> i32 {
    let parser = match p {
        Some(val) => val,
        None => return -1,
    };

    let mut quoted = parser.quoted;
    let mut pstate = parser.pstate;
    let mut spaces = parser.spaces;
    let mut entry_pos = parser.entry_pos;

    if pstate == FIELD_BEGUN && parser.quoted != 0 && (parser.options & CSV_STRICT) != 0 && (parser.options & CSV_STRICT_FINI) != 0 {
        parser.status = CSV_EPARSE;
        return -1;
    }

    match pstate {
        FIELD_MIGHT_HAVE_ENDED => {
            if parser.entry_pos >= parser.spaces + 1 {
                parser.entry_pos -= parser.spaces + 1;
            } else {
                parser.entry_pos = 0;
            }
            entry_pos = parser.entry_pos;
            let _ = quoted;
            let _ = spaces;
            
            if quoted == 0 {
                if parser.entry_pos < parser.entry_buf.len() {
                    parser.entry_buf[parser.entry_pos] = 0;
                }
            } else {
                if parser.entry_pos < parser.entry_buf.len() && (parser.options & CSV_APPEND_NULL) != 0 {
                    parser.entry_buf[parser.entry_pos] = 0;
                }
            }

            if let Some(ref mut c1) = cb1 {
                if (parser.options & CSV_EMPTY_IS_NULL) != 0 && quoted == 0 && entry_pos == 0 {
                    c1(&[], 0, data);
                } else {
                    let slice = if entry_pos <= parser.entry_buf.len() {
                        &parser.entry_buf[..entry_pos]
                    } else {
                        &parser.entry_buf[..]
                    };
                    c1(slice, entry_pos, data);
                }
            }
            pstate = FIELD_NOT_BEGUN;
            entry_pos = 0;
            quoted = 0;
            spaces = 0;

            if let Some(ref mut c2) = cb2 {
                c2(-1, data);
            }
            pstate = ROW_NOT_BEGUN;
            entry_pos = 0;
            quoted = 0;
            spaces = 0;
        }
        FIELD_NOT_BEGUN | FIELD_BEGUN => {
            if quoted == 0 {
                if parser.entry_pos < parser.entry_buf.len() {
                    parser.entry_buf[parser.entry_pos] = 0;
                }
            } else {
                if parser.entry_pos < parser.entry_buf.len() && (parser.options & CSV_APPEND_NULL) != 0 {
                    parser.entry_buf[parser.entry_pos] = 0;
                }
            }

            if let Some(ref mut c1) = cb1 {
                if (parser.options & CSV_EMPTY_IS_NULL) != 0 && quoted == 0 && entry_pos == 0 {
                    c1(&[], 0, data);
                } else {
                    let slice = if entry_pos <= parser.entry_buf.len() {
                        &parser.entry_buf[..entry_pos]
                    } else {
                        &parser.entry_buf[..]
                    };
                    c1(slice, entry_pos, data);
                }
            }
            pstate = FIELD_NOT_BEGUN;
            entry_pos = 0;
            quoted = 0;
            spaces = 0;

            if let Some(ref mut c2) = cb2 {
                c2(-1, data);
            }
            pstate = ROW_NOT_BEGUN;
            entry_pos = 0;
            quoted = 0;
            spaces = 0;
        }
        ROW_NOT_BEGUN => {}
        _ => {}
    }

    parser.spaces = 0;
    parser.quoted = 0;
    parser.entry_pos = 0;
    parser.status = 0;
    parser.pstate = ROW_NOT_BEGUN;

    0
}

pub fn csv_get_buffer_size(p: Option<&CsvParser>) -> usize {
    match p {
        Some(parser) => parser.entry_size,
        None => 0,
    }
}

fn csv_increase_buffer(p: &mut CsvParser) -> i32 {
    let mut to_add = p.blk_size;

    if p.entry_size >= usize::MAX - to_add {
        to_add = usize::MAX - p.entry_size;
    }

    if to_add == 0 {
        p.status = CSV_ETOOBIG;
        return -1;
    }

    loop {
        let new_len = match p.entry_size.checked_add(to_add) {
            Some(l) => l,
            None => {
                to_add /= 2;
                if to_add == 0 {
                    p.status = CSV_ENOMEM;
                    return -1;
                }
                continue;
            }
        };

        if let Some(_test) = p.entry_buf.try_reserve_exact(new_len - p.entry_buf.len()) {
            p.entry_buf.resize(new_len, 0);
            p.entry_size += to_add;
            return 0;
        } else {
            to_add /= 2;
            if to_add == 0 {
                p.status = CSV_ENOMEM;
                return -1;
            }
        }
    }
}

pub fn csv_parse(
    p: &mut CsvParser,
    s: Option<&[u8]>,
    mut cb1: Option<impl FnMut(&[u8], usize, *mut std::ffi::c_void)>,
    mut cb2: Option<impl FnMut(i32, *mut std::ffi::c_void)>,
    data: *mut std::ffi::c_void,
) -> usize {
    let us = match s {
        Some(slice) => slice,
        None => return 0,
    };

    let len = us.len();
    let mut pos = 0;

    let delim = p.delim_char;
    let quote = p.quote_char;
    let is_space = p.is_space;
    let is_term = p.is_term;
    let mut quoted = p.quoted;
    let mut pstate = p.pstate;
    let mut spaces = p.spaces;
    let mut entry_pos = p.entry_pos;

    if p.entry_buf.is_empty() && len > 0 {
        if csv_increase_buffer(p) != 0 {
            p.quoted = quoted;
            p.pstate = pstate;
            p.spaces = spaces;
            p.entry_pos = entry_pos;
            return pos;
        }
    }

    while pos < len {
        let limit = if (p.options & CSV_APPEND_NULL) != 0 {
            if p.entry_size > 0 { p.entry_size - 1 } else { 0 }
        } else {
            p.entry_size
        };

        if entry_pos == limit {
            if csv_increase_buffer(p) != 0 {
                p.quoted = quoted;
                p.pstate = pstate;
                p.spaces = spaces;
                p.entry_pos = entry_pos;
                return pos;
            }
        }

        let c = us[pos];
        pos += 1;

        macro_rules! submit_field {
            () => {
                if quoted == 0 {
                    if spaces <= entry_pos {
                        entry_pos -= spaces;
                    } else {
                        entry_pos = 0;
                    }
                }
                if (p.options & CSV_APPEND_NULL) != 0 {
                    if entry_pos < p.entry_buf.len() {
                        p.entry_buf[entry_pos] = 0;
                    }
                }
                if let Some(ref mut c1) = cb1 {
                    if (p.options & CSV_EMPTY_IS_NULL) != 0 && quoted == 0 && entry_pos == 0 {
                        c1(&[], 0, data);
                    } else {
                        let slice = if entry_pos <= p.entry_buf.len() {
                            &p.entry_buf[..entry_pos]
                        } else {
                            &p.entry_buf[..]
                        };
                        c1(slice, entry_pos, data);
                    }
                }
                pstate = FIELD_NOT_BEGUN;
                entry_pos = 0;
                quoted = 0;
                spaces = 0;
            };
        }

        macro_rules! submit_row {
            ($chr:expr) => {
                if let Some(ref mut c2) = cb2 {
                    c2($chr, data);
                }
                pstate = ROW_NOT_BEGUN;
                entry_pos = 0;
                quoted = 0;
                spaces = 0;
            };
        }

        macro_rules! submit_char {
            ($chr:expr) => {
                if entry_pos < p.entry_buf.len() {
                    p.entry_buf[entry_pos] = $chr;
                }
                entry_pos += 1;
            };
        }

        let is_sp = match is_space {
            Some(f) => f(c) != 0,
            None => c == CSV_SPACE || c == CSV_TAB,
        };

        let is_tm = match is_term {
            Some(f) => f(c) != 0,
            None => c == CSV_CR || c == CSV_LF,
        };

        switch_block: loop {
            match pstate {
                ROW_NOT_BEGUN | FIELD_NOT_BEGUN => {
                    if is_sp && c != delim {
                        break switch_block;
                    } else if is_tm {
                        if pstate == FIELD_NOT_BEGUN {
                            submit_field!();
                            submit_row!(c as i32);
                        } else {
                            if (p.options & CSV_REPALL_NL) != 0 {
                                submit_row!(c as i32);
                            }
                        }
                        break switch_block;
                    } else if c == delim {
                        submit_field!();
                        break switch_block;
                    } else if c == quote {
                        pstate = FIELD_BEGUN;
                        quoted = 1;
                        break switch_block;
                    } else {
                        pstate = FIELD_BEGUN;
                        quoted = 0;
                        submit_char!(c);
                        break switch_block;
                    }
                }
                FIELD_BEGUN => {
                    if c == quote {
                        if quoted != 0 {
                            submit_char!(c);
                            pstate = FIELD_MIGHT_HAVE_ENDED;
                        } else {
                            if (p.options & CSV_STRICT) != 0 {
                                p.status = CSV_EPARSE;
                                p.quoted = quoted;
                                p.pstate = pstate;
                                p.spaces = spaces;
                                p.entry_pos = entry_pos;
                                return pos - 1;
                            }
                            submit_char!(c);
                            spaces = 0;
                        }
                    } else if c == delim {
                        if quoted != 0 {
                            submit_char!(c);
                        } else {
                            submit_field!();
                        }
                    } else if is_tm {
                        if quoted == 0 {
                            submit_field!();
                            submit_row!(c as i32);
                        } else {
                            submit_char!(c);
                        }
                    } else if quoted == 0 && is_sp {
                        submit_char!(c);
                        spaces += 1;
                    } else {
                        submit_char!(c);
                        spaces = 0;
                    }
                    break switch_block;
                }
                FIELD_MIGHT_HAVE_ENDED => {
                    if c == delim {
                        let total_deduct = spaces + 1;
                        if entry_pos >= total_deduct {
                            entry_pos -= total_deduct;
                        } else {
                            entry_pos = 0;
                        }
                        submit_field!();
                    } else if is_tm {
                        let total_deduct = spaces + 1;
                        if entry_pos >= total_deduct {
                            entry_pos -= total_deduct;
                        } else {
                            entry_pos = 0;
                        }
                        submit_field!();
                        submit_row!(c as i32);
                    } else if is_sp {
                        submit_char!(c);
                        spaces += 1;
                    } else if c == quote {
                        if spaces > 0 {
                            if (p.options & CSV_STRICT) != 0 {
                                p.status = CSV_EPARSE;
                                p.quoted = quoted;
                                p.pstate = pstate;
                                p.spaces = spaces;
                                p.entry_pos = entry_pos;
                                return pos - 1;
                            }
                            spaces = 0;
                            submit_char!(c);
                        } else {
                            pstate = FIELD_BEGUN;
                        }
                    } else {
                        if (p.options & CSV_STRICT) != 0 {
                            p.status = CSV_EPARSE;
                            p.quoted = quoted;
                            p.pstate = pstate;
                            p.spaces = spaces;
                            p.entry_pos = entry_pos;
                            return pos - 1;
                        }
                        pstate = FIELD_BEGUN;
                        spaces = 0;
                        submit_char!(c);
                    }
                    break switch_block;
                }
                _ => {
                    break switch_block;
                }
            }
        }
    }

    p.quoted = quoted;
    p.pstate = pstate;
    p.spaces = spaces;
    p.entry_pos = entry_pos;
    pos
}

pub fn csv_write(dest: &mut [u8], src: &[u8]) -> usize {
    csv_write2(dest, src, CSV_QUOTE)
}

pub fn csv_write2(dest: &mut [u8], src: &[u8], quote: u8) -> usize {
    let mut chars: usize = 0;
    let mut dest_idx = 0;
    let dest_size = dest.len();

    if dest_size > 0 {
        if dest_idx < dest_size {
            dest[dest_idx] = quote;
            dest_idx += 1;
        }
    }
    if chars < usize::MAX {
        chars += 1;
    }

    for &c in src {
        if c == quote {
            if dest_size > chars {
                if dest_idx < dest_size {
                    dest[dest_idx] = quote;
                    dest_idx += 1;
                }
            }
            if chars < usize::MAX {
                chars += 1;
            }
        }
        if dest_size > chars {
            if dest_idx < dest_size {
                dest[dest_idx] = c;
                dest_idx += 1;
            }
        }
        if chars < usize::MAX {
            chars += 1;
        }
    }

    if dest_size > chars {
        if dest_idx < dest_size {
            dest[dest_idx] = quote;
        }
    }
    if chars < usize::MAX {
        chars += 1;
    }

    chars
}
